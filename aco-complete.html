Auth System — apps/api/src/plugins/auth.ts

apps/api/src/plugins/auth.tsTypeScript
import fp from 'fastify-plugin'
import { createHash, randomBytes } from 'crypto'
import bcrypt from 'bcryptjs'
import { authenticator } from 'otplib'
import { db, apiKeys, users, tenants } from '@aco/db'

// Every request gets tenantId + userId attached before route handlers run
declare module 'fastify' {
  interface FastifyRequest {
    tenantId: string
    userId:   string
    userRole: string
  }
}

export default fp(async function authPlugin(app) {

  // ── JWT auth (web dashboard + Discord bot internal calls) ─────
  app.decorateRequest('tenantId', '')
  app.decorateRequest('userId', '')

  app.addHook('preHandler', async (req, reply) => {
    const authHeader = req.headers.authorization

    if (authHeader?.startsWith('Bearer ')) {
      const token = authHeader.slice(7)

      // Try API key first (starts with 'aco_')
      if (token.startsWith('aco_')) {
        const hashed = createHash('sha256').update(token).digest('hex')
        const key = await db.query.apiKeys.findFirst({
          where: (k, { eq }) => eq(k.hashedKey, hashed)
        })
        if (!key) return reply.code(401).send({ error: 'Invalid API key' })
        req.tenantId = key.tenantId
        req.userId   = 'api'
        req.userRole = 'api'
        return
      }

      // JWT token
      try {
        const payload = await app.jwt.verify(token)
        req.tenantId = payload.tenantId
        req.userId   = payload.sub
        req.userRole = payload.role
      } catch {
        return reply.code(401).send({ error: 'Invalid token' })
      }
    }
  })

  // ── Helpers ───────────────────────────────────────────────────
  app.decorate('hashPassword',  (pw: string) => bcrypt.hash(pw, 12))
  app.decorate('checkPassword', (pw: string, hash: string) => bcrypt.compare(pw, hash))
  app.decorate('generateApiKey', () => `aco_${randomBytes(32).toString('hex')}`)
  app.decorate('hashApiKey', (k: string) => createHash('sha256').update(k).digest('hex'))
  app.decorate('generateTotp', () => authenticator.generateSecret())
  app.decorate('verifyTotp',   (token: string, secret: string) =>
    authenticator.verify({ token, secret }))
})

import { FastifyPluginAsync } from 'fastify'
import { z } from 'zod'
import { enqueueCheckout } from '@aco/queue'
import { checkUsageCap } from '../services/usage'
import { db, checkoutTasks, checkoutEvents } from '@aco/db'

const CreateTaskSchema = z.object({
  profileId: z.string().uuid(),
  retailer:  z.enum(['dominos','shake-shack','wingstop' /* … all 14 */]),
  items:     z.array(z.object({
    id:       z.string(),
    name:     z.string(),
    quantity: z.number().int().positive(),
    options:  z.record(z.string()).optional(),
  })).min(1),
})

const tasks: FastifyPluginAsync = async (app) => {

  // POST /v1/tasks — create checkout task
  app.post('/', async (req, reply) => {
    const body = CreateTaskSchema.parse(req.body)
    await checkUsageCap(req.tenantId)         // throws 429 if over limit
    const [task] = await db.insert(checkoutTasks).values({
      tenantId:  req.tenantId,
      userId:    req.userId,
      profileId: body.profileId,
      retailer:  body.retailer,
      items:     body.items,
    }).returning()
    await enqueueCheckout({ taskId: task.id, ...body })
    return reply.code(201).send(task)
  })

  // GET /v1/tasks/:id — task + full event log
  app.get('/:id', async (req, reply) => {
    const task = await db.query.checkoutTasks.findFirst({
      where: (t, { eq, and }) => and(
        eq(t.id, (req.params as any).id),
        eq(t.tenantId, req.tenantId)        // ALWAYS scope by tenant
      ),
      with: { events: { orderBy: (e, { asc }) => [asc(e.ts)] } }
    })
    if (!task) return reply.code(404).send()
    return task
  })

  // GET /v1/tasks/:id/stream — SSE live state updates
  app.get('/:id/stream', async (req, reply) => {
    reply.raw.setHeader('Content-Type', 'text/event-stream')
    reply.raw.setHeader('Cache-Control', 'no-cache')
    const send = (data: object) =>
      reply.raw.write(`data: ${JSON.stringify(data)}\n\n`)
    // Subscribe to Redis pub/sub channel for this task
    const sub = await subscribeToTask((req.params as any).id, send)
    req.raw.on('close', () => sub.unsubscribe())
  })
}

export interface ProxyRecord {
  host: string; port: number
  username?: string; password?: string
  protocol: 'http' | 'https' | 'socks5'
}

// Handles all common proxy formats:
// host:port
// host:port:user:pass
// user:pass@host:port
// protocol://host:port
// protocol://user:pass@host:port
export function parseProxy(raw: string): ProxyRecord {
  const normalised = raw.trim()
  const withProto = normalised.includes('://')
    ? normalised
    : `http://${normalised}`

  try {
    const url = new URL(withProto)
    return {
      host:     url.hostname,
      port:     parseInt(url.port),
      username: url.username || undefined,
      password: url.password || undefined,
      protocol: (url.protocol.replace(':', '') || 'http') as ProxyRecord['protocol'],
    }
  } catch {
    // host:port:user:pass format
    const [host, portStr, username, password] = raw.split(':')
    return { host, port: parseInt(portStr), username, password, protocol: 'http' }
  }
}

export function toPlaywrightProxy(p: ProxyRecord) {
  return {
    server:   `${p.protocol}://${p.host}:${p.port}`,
    username: p.username,
    password: p.password,
  }
}

    await injectToken(page, type, token)
    return 'solved'

import { createCipheriv, createDecipheriv, randomBytes } from 'crypto'

const KEY = Buffer.from(process.env.CARD_ENCRYPTION_KEY!, 'hex')  // 32 bytes

export function encryptField(plaintext: string): string {
  const iv      = randomBytes(12)
  const cipher  = createCipheriv('aes-256-gcm', KEY, iv)
  const enc     = Buffer.concat([cipher.update(plaintext, 'utf8'), cipher.final()])
  const authTag = cipher.getAuthTag()
  // Format: iv_hex:ciphertext_hex:authtag_hex
  return [iv.toString('hex'), enc.toString('hex'), authTag.toString('hex')].join(':')
}

export function decryptField(encrypted: string): string {
  const [ivHex, encHex, tagHex] = encrypted.split(':')
  const decipher = createDecipheriv('aes-256-gcm', KEY, Buffer.from(ivHex, 'hex'))
  decipher.setAuthTag(Buffer.from(tagHex, 'hex'))
  return decipher.update(Buffer.from(encHex, 'hex')).toString() + decipher.final().toString()
}

// Decrypted PAN lives only inside this function call — never assigned to a variable
// that gets logged or passed to non-checkout code
export function withDecryptedCard<T>(

  fn: (pan: string, cvv: string) => Promise<T>

Uber Eats — Group Order Flow

The confirmed implementation approach from the client changes this significantly from a standard aggregator integration. The customer builds their own cart on Uber Eats, creates a group order, and sends the group order link to the ACO. The ACO joins via that link — the cart, address, restaurant, and items are all already there. The ACO's job is purely to complete the checkout, not to build the cart.

Confirmed Uber Eats group order flow
Customer builds cart on Uber Eats → Customer creates group order → Customer sends link to ACO
ACO receives group order URL → ACO navigates to link → Join group order (cart pre-populated)
Review cart (no changes needed) → Proceed to checkout → CAPTCHA if triggered
Enter payment → Submit with human-paced timing → Order confirmed → extract order ID
What this eliminates: No item search, no store selection, no address input, no menu navigation. The customer has already done all of that. The adapter only needs to handle: joining the group order URL, reviewing the pre-built cart, entering payment, and submitting. This is substantially simpler than a full aggregator integration.

export class UberEatsAdapter extends RetailerAdapter {
  readonly id          = 'uber-eats'
  readonly displayName = 'Uber Eats'
  readonly baseUrl     = 'https://eats.uber.com'

  // Conservative — Uber has account-level behaviour analysis
  readonly rateLimit = {
    requestsPerMinute:        10,
    concurrentMax:            3,
    minDelayBetweenActionsMs: 1500,  // longer delays — humanisation is critical here
  }

  async buildCart(items: CartItem[], s: CheckoutSession) {
    // Cart is pre-built by the customer via group order.
    // The group order URL is passed in via s.metadata.groupOrderUrl.
    // This step just navigates to the link and joins.
    const groupOrderUrl = s.metadata?.groupOrderUrl
    if (!groupOrderUrl) throw new FatalError('Uber Eats: no group order URL provided')
    if (!UBER_GROUP_ORDER_PATTERN.test(groupOrderUrl)) {
      throw new FatalError('Uber Eats: invalid group order URL format')
    }

    await logEvent(s.taskId, 'CART_BUILDING', { groupOrderUrl })
    await s.page.goto(groupOrderUrl, { waitUntil: 'networkidle' })
    await this.humanDelay(1200, 2400)  // realistic page-load pause

    // Handle any login prompt if not already signed in
    const loginPrompt = await s.page.$(SELECTORS.loginPrompt)
    if (loginPrompt) {
      await ensureUberSession(s.page, s.context, s.credentials, s.taskId, s.redis)
      await s.page.goto(groupOrderUrl, { waitUntil: 'networkidle' })
      await this.humanDelay(1000, 2000)
    }

    // Join the group order
    await s.page.waitForSelector(SELECTORS.joinGroupOrder, { timeout: 15000 })
    await this.humanDelay(800, 1600)
    await s.page.click(SELECTORS.joinGroupOrder)
    await s.page.waitForLoadState('networkidle')

    // Verify the cart has items — it should, since customer pre-built it
    const cartItems = await s.page.$$(SELECTORS.cartItemRow)
    if (cartItems.length === 0) {
      throw new FatalError('Uber Eats: group order cart is empty after joining')
    }
    log.info({ taskId: s.taskId, itemCount: cartItems.length }, 'Uber Eats: joined group order, cart populated')
  }

  async setAddress(_addr: Address, _s: CheckoutSession) {
    // Address is embedded in the group order — no action needed here
  }

  async selectPayment(method: PaymentMethod, s: CheckoutSession) {
    s.paymentMethod = method  // stored, applied in submitCheckout
  }

  async submitCheckout(s: CheckoutSession): Promise<CheckoutResult> {
    // Navigate to checkout from the group order cart
    await s.page.waitForSelector(SELECTORS.proceedToCheckout, { state: 'visible' })
    await this.humanDelay(1000, 2200)  // pause — don't rush straight to checkout
    await s.page.click(SELECTORS.proceedToCheckout)
    await s.page.waitForLoadState('networkidle')
    await this.humanDelay(800, 1500)

    // Enter payment — Uber Eats card fields are typically in an iframe
    await enterUberPayment(s.page, s.paymentMethod!, s.taskId)

    // ── HUMANISATION: critical on Uber Eats ─────────────────────────────
    // Variable pause before submit — 2–5 seconds, randomised.
    // Uber's behaviour analysis looks for machine-speed form fill → immediate submit.
    // A human pauses, maybe scrolls to review the order, then places it.
    await humanReviewPause(s.page)

    // Submit
    await s.page.waitForSelector(SELECTORS.placeOrder, { state: 'visible' })
    await s.page.click(SELECTORS.placeOrder)

    // Race outcomes
    const outcome = await Promise.race([
      s.page.waitForURL(/order-status|order\/[a-f0-9-]+/, { timeout: 30000 }).then(() => 'confirmed'),
      s.page.waitForSelector(SELECTORS.paymentError, { timeout: 30000 }).then(() => 'declined'),
      s.page.waitForSelector(SELECTORS.captchaContainer, { timeout: 30000 }).then(() => 'captcha'),
    ])

    if (outcome === 'captcha')  return { success: false, requiresVerification: 'CAPTCHA' }
    if (outcome === 'declined') return { success: false, failureReason: 'PAYMENT_DECLINED' }

    // Extract order ID
    const orderId = await extractUberOrderId(s.page)
    await extendUberSession(s.credentials.email, s.redis)
    log.info({ taskId: s.taskId, orderId }, 'Uber Eats: order confirmed')
    return { success: true, orderId }
  }

  async detectVerification(s: CheckoutSession) {
    return await s.page.$(SELECTORS.captchaContainer) ? 'captcha' : null
  }

  async isRateLimited(s: CheckoutSession) {
    const url = s.page.url()
    return url.includes('rate-limit') || url.includes('too-many')
  }

  async isSessionValid(s: CheckoutSession) {
    return !!(await s.page.$(SELECTORS.accountAvatar).catch(() => null))
  }
}

// ── Humanisation helper ──────────────────────────────────────────────────
// Simulates a human reviewing the order before submitting.
// Scrolls down slowly, then back up, then pauses — variable timing.
async function humanReviewPause(page: Page) {
  // Scroll down to review items
  await page.evaluate(() => window.scrollBy({ top: 300, behavior: 'smooth' }))
  await sleep(600 + Math.random() * 800)
  // Scroll back up to the submit button
  await page.evaluate(() => window.scrollBy({ top: -200, behavior: 'smooth' }))
  await sleep(800 + Math.random() * 1200)
  // Final pause before clicking — 1.5 to 4 seconds, highly variable
  await sleep(1500 + Math.random() * 2500)
}
Humanisation note: Uber Eats performs account-level behaviour analysis. The biggest signals are: form fill-to-submit speed (humans take time to review), scroll patterns (humans scroll down to check the order), and mouse movement between fields. The humanReviewPause() function above addresses the most critical one — the pause before submission. Mouse movement and realistic typing delays are handled by the base class humanDelay() and typeHuman() helpers. The client specifically flagged this as a priority — take it seriously during testing and adjust timing if Uber starts flagging sessions.

Uber Eats build checklist

  • URL validation — Regex validates group order URL format before task is accepted. Invalid URLs rejected immediately with clear error.
  • Session management — Uber account login persisted in Redis (same pattern as Instacart). Re-login on session expiry before navigating to group order URL.
  • Group order join flow — Navigate to URL → handle any login prompt → click join → verify cart is not empty.
  • setAddress() is a no-op — Address comes from the group order. No address input step in this adapter.
  • Payment entry — Detect if card fields are in an iframe (Uber uses Braintree/Stripe hosted fields). withDecryptedCard() pattern, never log PAN.
  • humanReviewPause() — Scroll + variable pause before submit. Test timing against Uber's detection. Adjust if sessions get flagged.
  • Submit race — Race confirmed/declined/captcha. CAPTCHA → pipeline. Declined → card cycling.
  • Order ID extraction — Extract from confirmation URL pattern or page element. Log warning if page reached but ID not found (order likely went through — do not resubmit).
  • Discord command — Conditional group_order_url option when uber-eats selected. Clear error message on bad URL.
  • API schema — items[] accepted as empty for uber-eats tasks. metadata.groupOrderUrl required and validated.
  • E2E test — Full flow test with a real group order link against a test account. Verify humanisation timing doesn't trigger detection.
  • Stealth test — Run session against bot.sannysoft.com before touching real Uber Eats. Uber's detection is sophisticated.

import { RetailerAdapter, CartItem, CheckoutResult } from '../core/adapter'

export class DominosAdapter extends RetailerAdapter {
  readonly id          = 'dominos'
  readonly displayName = "Domino's"
  readonly baseUrl     = 'https://www.dominos.com'
  readonly rateLimit   = { requestsPerMinute: 20, concurrentMax: 3, minDelayBetweenActionsMs: 800 }

  async buildCart(items: CartItem[], s: CheckoutSession) {
    await s.page.goto(`${this.baseUrl}/en/`)
    await this.humanDelay()

    for (const item of items) {
      await this.click(s.page, `[data-product-code="${item.id}"]`)
      await this.humanDelay()

      // Apply customisations
      if (item.options) {
        for (const [key, value] of Object.entries(item.options)) {
          await this.click(s.page, `[data-option="${key}"][data-value="${value}"]`)
          await this.humanDelay(300, 700)
        }
      }

      await this.click(s.page, '[data-testid="add-to-cart"]')
      await this.humanDelay(500, 1200)
    }
  }

  async setAddress(addr: Address, s: CheckoutSession) {
    await this.click(s.page, '[data-testid="delivery-address"]')
    await this.type(s.page, '#address-line1', addr.line1)
    if (addr.line2) await this.type(s.page, '#address-line2', addr.line2)
    await this.type(s.page, '#city', addr.city)
    await s.page.selectOption('#state', addr.state)
    await this.type(s.page, '#zip', addr.zip)
    await this.click(s.page, '[data-testid="address-submit"]')
    await this.humanDelay()
  }

  async selectPayment(method: PaymentMethod, s: CheckoutSession) {
    await this.click(s.page, '[data-testid="payment-card"]')
    // PAN is decrypted inside withDecryptedCard — never stored in a variable outside
    await withDecryptedCard(method, async (pan, cvv) => {
      await this.type(s.page, '#card-number', pan)
      await this.type(s.page, '#card-expiry', `${method.expMonth}/${method.expYear}`)
      await this.type(s.page, '#card-cvv', cvv)
    })
    if (method.billingZip) await this.type(s.page, '#billing-zip', method.billingZip)
  }

  async submitCheckout(s: CheckoutSession): Promise<CheckoutResult> {
    await this.click(s.page, '[data-testid="place-order"]')

    // Race: success confirmation vs error states
    const result = await Promise.race([
      s.page.waitForURL(/order-confirmation/, { timeout: 20000 })
        .then(() => 'success'),
      s.page.waitForSelector('[data-testid="payment-error"]', { timeout: 20000 })
        .then(() => 'declined'),
      s.page.waitForSelector('[data-testid="captcha-container"]', { timeout: 20000 })
        .then(() => 'captcha'),
    ])

    if (result === 'success') {
      const orderId = await s.page.textContent('[data-testid="order-id"]')
      return { success: true, orderId: orderId?.trim() }
    }
    if (result === 'declined')
      return { success: false, failureReason: 'PAYMENT_DECLINED' }
    return { success: false, requiresVerification: 'CAPTCHA' }
  }

  async detectVerification(s: CheckoutSession) {
    return await s.page.$('[data-sitekey]') ? 'recaptcha-v2' : null
  }
  async isRateLimited(s: CheckoutSession) {
    return s.page.url().includes('sorry') || s.page.url().includes('block')
  }
  async isSessionValid(s: CheckoutSession) {
    return !s.page.url().includes('timeout')
  }
}

Security Checklist

  • Card encryption — AES-256-GCM with per-value random IV. Key only in Doppler. Never in code, logs, or stack traces
  • ESLint no-card-log rule — Custom rule that flags any log.* / console.* call with args containing 'pan', 'cvv', 'card_number', or 'encryption_key'
  • Tenant isolation test — Integration test: User A cannot access Tenant B's resources via any API endpoint, even with valid JWT
  • API key entropy — 32 random bytes (256 bits), prefixed aco_. Store only SHA-256 hash in DB
  • JWT short-lived — 7 day expiry, refresh token pattern for the web dashboard
  • Webhook HMAC — Sign payloads with HMAC-SHA256. Document verification for customers
  • Rate limiting — Per-API-key sliding window in Redis. Configurable per key by reseller
  • SQL injection — Drizzle uses parameterised queries throughout. Never string-concatenate into SQL
  • SSRF prevention — Webhook URLs validated against allowlist of protocols (https only); reject private IP ranges
  • Redis ACLs — Separate Redis users for queue vs cache vs pubsub with minimal permissions
  • Worker isolation — Each worker is a separate child process; a compromised worker cannot access other tenants' sessions
  • Dependencies — npm audit in CI pipeline; Dependabot PRs auto-opened for security patches
  • Secrets rotation — Document key rotation procedure for CARD_ENCRYPTION_KEY (re-encrypt all cards on rotation)
